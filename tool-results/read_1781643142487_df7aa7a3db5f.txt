   768→   768→-- Lead score calculation
   769→   769→CREATE OR REPLACE FUNCTION public.calculate_lead_score(lead_row public.leads)
   770→   770→RETURNS INT
   771→   771→LANGUAGE plpgsql
   772→   772→AS $$
   773→   773→DECLARE
   774→   774→  score INT := 0;
   775→   775→BEGIN
   776→   776→  IF lead_row.source = 'referral' THEN score := score + 30;
   777→   777→  ELSIF lead_row.source = 'google_ads' THEN score := score + 20;
   778→   778→  ELSIF lead_row.source = 'website' THEN score := score + 15;
   779→   779→  ELSIF lead_row.source = 'walk_in' THEN score := score + 25;
   780→   780→  ELSE score := score + 10;
   781→   781→  END IF;
   782→   782→
   783→   783→  IF lead_row.estimated_value IS NOT NULL THEN
   784→   784→    IF lead_row.estimated_value >= 100000 THEN score := score + 25;
   785→   785→    ELSIF lead_row.estimated_value >= 50000 THEN score := score + 20;
   786→   786→    ELSIF lead_row.estimated_value >= 10000 THEN score := score + 10;
   787→   787→    END IF;
   788→   788→  END IF;
   789→   789→
   790→   790→  IF lead_row.case_type IN ('corporate','property') THEN score := score + 15;
   791→   791→  ELSIF lead_row.case_type = 'family' THEN score := score + 10;
   792→   792→  END IF;
   793→   793→
   794→   794→  IF lead_row.status = 'qualified' THEN score := score + 15;
   795→   795→  ELSIF lead_row.status = 'consultation_scheduled' THEN score := score + 20;
   796→   796→  END IF;
   797→   797→
   798→   798→  RETURN LEAST(score, 100);
   799→   799→END;
   800→   800→$$;
