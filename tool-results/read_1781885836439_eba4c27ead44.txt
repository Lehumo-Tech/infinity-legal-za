    69→    69→-- Auto-update updated_at
    70→    70→CREATE OR REPLACE FUNCTION public.handle_updated_at()
    71→    71→RETURNS TRIGGER AS $$
    72→    72→BEGIN
    73→    73→  NEW.updated_at = now();
    74→    74→  RETURN NEW;
    75→    75→END;
    76→    76→$$ LANGUAGE plpgsql;
