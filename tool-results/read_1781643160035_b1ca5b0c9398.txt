  1281→CREATE OR REPLACE VIEW public.v_crm_dashboard AS
  1282→SELECT
  1283→  (SELECT count(*) FROM public.leads WHERE status = 'new') AS new_leads_count,
  1284→  (SELECT count(*) FROM public.leads WHERE status = 'qualified') AS qualified_leads_count,
  1285→  (SELECT count(*) FROM public.cases WHERE status = 'active') AS active_cases_count,
  1286→  (SELECT count(*) FROM public.cases WHERE status = 'intake') AS intake_cases_count,
  1287→  (SELECT count(*) FROM public.intake_submissions WHERE status = 'submitted') AS pending_intakes_count,
  1288→  (SELECT count(*) FROM public.consultations WHERE status = 'scheduled') AS scheduled_consultations_count,
  1289→  (SELECT COALESCE(sum(estimated_value), 0) FROM public.leads WHERE status IN ('new','contacted','qualified')) AS pipeline_value,
  1290→  (SELECT COALESCE(sum(retainer_amount), 0) FROM public.cases WHERE status = 'active') AS active_case_value,
  1291→  (SELECT count(*) FROM public.ai_analyses WHERE status = 'processing') AS ai_processing_count,
  1292→  (SELECT count(*) FROM public.crm_contact_messages WHERE status = 'unread') AS unread_messages_count;

  1294→CREATE OR REPLACE VIEW public.v_lead_pipeline AS
  1295→SELECT
  1296→  ls.name AS stage_name,
  1297→  ls.slug AS stage_slug,
  1298→  ls.sort_order,
  1299→  ls.color,
  1300→  count(l.id) AS lead_count,
  1301→  COALESCE(sum(l.estimated_value), 0) AS total_value
  1302→FROM public.lead_pipeline_stages ls
  1303→LEFT JOIN public.leads l ON l.status = ls.slug
  1304→GROUP BY ls.id, ls.name, ls.slug, ls.sort_order, ls.color
  1305→ORDER BY ls.sort_order;
