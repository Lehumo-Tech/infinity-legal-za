   109→function formatZAR(amount: number): string {
   110→  return new Intl.NumberFormat('en-ZA', {
   111→    style: 'currency',
   112→    currency: 'ZAR',
   113→    minimumFractionDigits: 0,
   114→    maximumFractionDigits: 0,
   115→  }).format(amount);
   116→}
   117→
   118→function formatSADate(dateStr: string): string {
   119→  try {
   120→    const d = new Date(dateStr);
   121→    return d.toLocaleDateString('en-ZA', { day: '2-digit', month: '2-digit', year: 'numeric' });
   122→  } catch {
   123→    return dateStr;
   124→  }
   125→}
   126→
   127→function formatTimeAgo(dateStr: string): string {
   128→  const now = Date.now();
   129→  const then = new Date(dateStr).getTime();
   130→  const diff = now - then;
   131→  const minutes = Math.floor(diff / 60000);
   132→  const hours = Math.floor(diff / 3600000);
   133→  const days = Math.floor(diff / 86400000);
   134→
   135→  if (minutes < 1) return 'Just now';
   136→  if (minutes < 60) return `${minutes}m ago`;
   137→  if (hours < 24) return `${hours}h ago`;
   138→  if (days < 7) return `${days}d ago`;
   139→  return formatSADate(dateStr);
   140→}
