    69→-- Auto-update updated_at
    70→CREATE OR REPLACE FUNCTION public.handle_updated_at()
    71→RETURNS TRIGGER AS $$
    72→BEGIN
    73→  NEW.updated_at = now();
    74→  RETURN NEW;
    75→END;
    76→$$ LANGUAGE plpgsql;
    77→
    78→CREATE TRIGGER set_profiles_updated_at
    79→  BEFORE UPDATE ON public.profiles
    80→  FOR EACH ROW EXECUTE FUNCTION public.handle_updated_at();

   754→-- ============================================
   755→-- FUNCTION: Auto-create profile on user signup
   756→-- ============================================
   757→
   758→CREATE OR REPLACE FUNCTION public.handle_new_user()
   759→RETURNS TRIGGER AS $$
   760→BEGIN
   761→  INSERT INTO public.profiles (user_id, email, full_name, role)
   762→  VALUES (
   763→    NEW.id,
   764→    NEW.email,
   765→    COALESCE(NEW.raw_user_meta_data->>'full_name', ''),
   766→    COALESCE((NEW.raw_user_meta_data->>'role')::public.user_role, 'client')
   767→  );
   768→  RETURN NEW;
   769→END;
   770→$$ LANGUAGE plpgsql SECURITY DEFINER;
