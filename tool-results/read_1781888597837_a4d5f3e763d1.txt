   190→        headers: { Authorization: `Bearer ${t}` },

   194→        setStats(data.data.stats);
   195→        setCharts(data.data.charts || null);
   196→        setFirmHealth(data.data.health || {});
   197→      }
   198→    } catch (e) {
   199→      console.error('Dashboard load error:', e);
   200→    }
   201→  };
   202→

   204→    if (!token) return;

   207→        headers: { Authorization: `Bearer ${token}` },

   217→    }

   323→      else if (currentView === 'documents') await loadDocuments();

   330→

   336→          await loadNotifications();

   338→      })();
   339→    }
   340→  }, [isAuthenticated, token]);

   513→      {/* Main content */}
