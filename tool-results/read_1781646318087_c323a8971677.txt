   279→-- ============================================
   280→-- 12. HELPER FUNCTION: is_admin()
   281→-- ============================================
   282→-- Returns true if the current authenticated user has an admin role
   283→
   284→CREATE OR REPLACE FUNCTION public.is_admin()
   285→RETURNS BOOLEAN AS $$
   286→BEGIN
   287→  RETURN EXISTS (
   288→    SELECT 1 FROM public.profiles
   289→    WHERE user_id = auth.uid()
   290→    AND role IN ('admin', 'managing_director', 'systems_admin')
   291→    AND is_active = true
   292→  );
   293→END;
   294→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   295→
   296→-- ============================================
   297→-- 13. HELPER FUNCTION: get_admin_role()
   298→-- ============================================
   299→-- Returns the admin role of the current user, or NULL
   300→
   301→CREATE OR REPLACE FUNCTION public.get_admin_role()
   302→RETURNS public.user_role AS $$
   303→DECLARE
   304→  user_role public.user_role;
   305→BEGIN
   306→  SELECT role INTO user_role
   307→  FROM public.profiles
   308→  WHERE user_id = auth.uid()
   309→  AND role IN ('admin', 'managing_director', 'systems_admin')
   310→  AND is_active = true;
   311→  
   312→  RETURN user_role;
   313→END;
   314→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

   594→-- View: User sign-up statistics
   595→CREATE OR REPLACE VIEW public.crm_signup_stats AS
   596→SELECT
   597→  DATE(p.created_at) AS signup_date,
   598→  COUNT(*) AS total_signups,
   599→  COUNT(*) FILTER (WHERE p.role = 'client') AS client_signups,
   600→  COUNT(*) FILTER (WHERE p.role = 'admin') AS admin_signups,
   601→  COUNT(*) FILTER (WHERE p.role IN ('associate', 'paralegal', 'candidate_attorney')) AS staff_signups,
   602→  COUNT(*) FILTER (WHERE p.is_active = true) AS active_users
   603→FROM public.profiles p
   604→GROUP BY DATE(p.created_at)
   605→ORDER BY signup_date DESC;
   606→
   607→-- View: Subscription revenue summary
   608→CREATE OR REPLACE VIEW public.crm_revenue_summary AS
   609→SELECT
   610→  DATE(pr.created_at) AS payment_date,
   611→  COUNT(*) AS total_payments,
   612→  SUM(pr.amount_gross) AS gross_revenue,
   613→  SUM(pr.amount_fee) AS total_fees,
   614→  SUM(pr.amount_net) AS net_revenue,
   615→  COUNT(*) FILTER (WHERE pr.payment_status = 'complete') AS completed_payments,
   616→  COUNT(*) FILTER (WHERE pr.payment_status = 'pending') AS pending_payments,
   617→  COUNT(*) FILTER (WHERE pr.payment_status = 'failed') AS failed_payments
   618→FROM public.payment_records pr
   619→GROUP BY DATE(pr.created_at)
   620→ORDER BY payment_date DESC;
   621→
   622→-- View: Lead conversion funnel
   623→CREATE OR REPLACE VIEW public.crm_lead_funnel AS
   624→SELECT
   625→  l.status,
   626→  COUNT(*) AS count,
   627→  l.source,
   628→  COUNT(*) FILTER (WHERE l.converted_case_id IS NOT NULL) AS converted_count
   629→FROM public.leads l
   630→GROUP BY l.status, l.source
   631→ORDER BY count DESC;
   632→
   633→-- View: Case pipeline overview
   634→CREATE OR REPLACE VIEW public.crm_case_pipeline AS
   635→SELECT
   636→  c.status,
   637→  c.case_type,
   638→  COUNT(*) AS case_count,
   639→  AVG(c.estimated_value) AS avg_value,
   640→  SUM(c.estimated_value) AS total_value,
   641→  COUNT(*) FILTER (WHERE c.is_high_risk = true) AS high_risk_count,
   642→  MIN(c.created_at) AS oldest_case,
   643→  MAX(c.created_at) AS newest_case
   644→FROM public.cases c
   645→GROUP BY c.status, c.case_type
   646→ORDER BY case_count DESC;
   647→
   648→-- View: Active subscription metrics
   649→CREATE OR REPLACE VIEW public.crm_subscription_metrics AS
   650→SELECT
   651→  us.status AS subscription_status,
   652→  pp.name AS plan_name,
   653→  pp.slug AS plan_slug,
   654→  COUNT(*) AS subscriber_count,
   655→  SUM(pp.price_monthly) AS monthly_recurring_revenue
   656→FROM public.user_subscriptions us
   657→JOIN public.pricing_plans pp ON us.plan_id = pp.id
   658→GROUP BY us.status, pp.name, pp.slug
   659→ORDER BY subscriber_count DESC;

   687→-- ============================================
   688→-- 20. TRIGGER: Log subscription events automatically
   689→-- ============================================
   690→
   691→CREATE OR REPLACE FUNCTION public.log_subscription_event()
   692→RETURNS TRIGGER AS $$
   693→BEGIN
   694→  INSERT INTO public.crm_subscription_events (
   695→    subscription_id, user_id, event_type, previous_status, new_status, plan_id
   696→  ) VALUES (
   697→    NEW.id,
   698→    NEW.user_id,
   699→    CASE
   700→      WHEN TG_OP = 'INSERT' THEN 'subscription_created'
   701→      WHEN TG_OP = 'UPDATE' AND OLD.status != NEW.status THEN 'status_changed'
   702→      WHEN TG_OP = 'UPDATE' AND OLD.cancel_at_period_end = false AND NEW.cancel_at_period_end = true THEN 'cancellation_requested'
   703→      WHEN TG_OP = 'UPDATE' THEN 'subscription_updated'
   704→    END,
   705→    OLD.status,
   706→    NEW.status,
   707→    NEW.plan_id
   708→  );
   709→  RETURN NEW;
   710→END;
   711→$$ LANGUAGE plpgsql SECURITY DEFINER;
   712→
   713→CREATE TRIGGER on_subscription_change
   714→  AFTER INSERT OR UPDATE ON public.user_subscriptions
   715→  FOR EACH ROW EXECUTE FUNCTION public.log_subscription_event();

   743→  END LOOP;
